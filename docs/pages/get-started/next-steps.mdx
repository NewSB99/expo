const VehicleCard = ({ vehicle, onReserve, onFavorite, chats, currentMessages, onSendMessage }) => (
  <Card className="p-4 shadow-lg rounded-2xl">
    <CardContent>
      <h2 className="text-xl font-semibold">{vehicle.model}</h2>
      {vehicle.images && vehicle.images.length > 0 && (

            <img key={index} src={img} alt={`${vehicle.model} ${index + 1}`} className="w-32 h-32 object-cover rounded" />
          ))}
        </div>
      )}
      <p>Prix: {vehicle.price} CHF / jour</p>
      <p>Conditions: {vehicle.conditions}</p>
      <p>Localisation: {vehicle.location}</p>

      <div className="mt-4">
        <Button onClick={() => onFavorite(vehicle)} className="bg-yellow-500">Ajouter aux favoris</Button>
        <Button onClick={() => onReserve(vehicle)} className="mt-2 w-full">
          {vehicle.available ? 'Réserver' : 'Indisponible'}
        </Button>
      </div>

      <div className="mt-4 p-2 border rounded-lg bg-gray-50">
        <h3 className="text-lg font-semibold">Discussion avec le loueur</h3>
        <div className="h-32 overflow-y-auto border-b mb-2 p-2">
          {(chats[vehicle.id] || []).map((msg, index) => (
            <p key={index} className="text-sm bg-gray-200 p-1 rounded mb-1">{msg}</p>
          ))}
        </div>
        <Input 
          type="text" 
          placeholder="Envoyer un message..." 
          value={currentMessages[vehicle.id] || ''} 
          onChange={(e) => onSendMessage(vehicle.id, e.target.value)} 
          className="mt-1"
        />
        <Button onClick={() => onSendMessage(vehicle.id)} className="mt-2">Envoyer</Button>
      </div>
    </CardContent>
  </Card>
);
